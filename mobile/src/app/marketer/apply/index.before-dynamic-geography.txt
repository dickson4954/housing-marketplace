import React, { useMemo, useState } from 'react';
import {
  Alert,
  Image,
  Pressable,
  ScrollView,
  StatusBar,
  StyleSheet,
  Text,
  TextInput,
  View,
} from 'react-native';
import { router } from 'expo-router';
import * as ImagePicker from 'expo-image-picker';
import {
  ArrowLeft,
  ArrowRight,
  Camera,
  Check,
  CheckCircle2,
  ImagePlus,
  MapPin,
  ShieldCheck,
  UserRound,
  X,
} from 'lucide-react-native';

import { useMarketers } from '@/context/MarketerContext';

const COLORS = {
  background: '#0B0F10',
  card: '#14191B',
  cardSoft: '#181E20',
  lime: '#D7FF5F',
  white: '#FFFFFF',
  muted: '#9CA5A7',
  border: '#252C2E',
  danger: '#FF6B6B',
};

type IdType = 'National ID' | 'Passport';

type ApplicationData = {
  firstName: string;
  lastName: string;
  phone: string;
  email: string;

  idType: IdType;
  idNumber: string;
  idFront: string | null;
  idBack: string | null;

  county: string;
  countyId: string;
  constituencyId: string;
  wardIds: string[];
  placeIds: string[];
  places: string[];
  timeInSelectedAreas: string;

  experience: 'Yes' | 'No' | '';
  experienceType: string;
  experienceLength: string;
  experienceDescription: string;

  occupation: string;
  workingWithAgency: 'Yes' | 'No' | '';
  agencyName: string;

  motivation: string;
  contribution: string;

  propertiesPerMonth: string;
  sourcingMethods: string[];

  knowsLandlords: 'Yes' | 'No' | '';
  landlordNetwork: string;

  comfortableCommunication: 'Yes' | 'No' | '';
  communicationMethods: string[];

  availability: string;
  followsProcedures: 'Yes' | 'No' | '';

  referenceName: string;
  referencePhone: string;
  referenceRelationship: string;

  declaration: boolean;
};

const INITIAL_DATA: ApplicationData = {
  firstName: '',
  lastName: '',
  phone: '',
  email: '',

  idType: 'National ID',
  idNumber: '',
  idFront: null,
  idBack: null,

  county: '',
  countyId: '',
  constituencyId: '',
  wardIds: [],
  placeIds: [],
  places: [],
  timeInSelectedAreas: '',

  experience: '',
  experienceType: '',
  experienceLength: '',
  experienceDescription: '',

  occupation: '',
  workingWithAgency: '',
  agencyName: '',

  motivation: '',
  contribution: '',

  propertiesPerMonth: '',
  sourcingMethods: [],

  knowsLandlords: '',
  landlordNetwork: '',

  comfortableCommunication: '',
  communicationMethods: [],

  availability: '',
  followsProcedures: '',

  referenceName: '',
  referencePhone: '',
  referenceRelationship: '',

  declaration: false,
};

const STEPS = [
  'Personal',
  'Identity',
  'Location',
  'Experience',
  'Review',
];

const PLACE_OPTIONS = [
  'Nairobi',
  'Kilimani',
  'Kileleshwa',
  'Westlands',
  'South B',
  'South C',
  'Kasarani',
  'Roysambu',
  'Meru Town',
  'Nchiru',
  'Kianjai',
  'Maua',
];

const TIME_IN_AREA_OPTIONS = [
  'Less than 1 year',
  '1–2 years',
  '3–5 years',
  'More than 5 years',
];

const EXPERIENCE_TYPES = [
  'Property marketing',
  'Property management',
  'Real-estate agency',
  'Property sourcing',
  'Property sales',
  'Rental management',
  'Other',
];

const EXPERIENCE_LENGTHS = [
  'Less than 1 year',
  '1–2 years',
  '3–5 years',
  'More than 5 years',
];

const OCCUPATIONS = [
  'Real-estate agent',
  'Self-employed',
  'Student',
  'Employed',
  'Business owner',
  'Property manager',
  'Other',
];

const PROPERTY_RANGES = [
  '1–5',
  '6–10',
  '11–20',
  'More than 20',
  'Not sure',
];

const SOURCING_METHODS = [
  'Local landlords',
  'Visiting properties',
  'Referrals',
  'Social media',
  'Real-estate networks',
  'Property agents',
  'Other',
];

const COMMUNICATION_METHODS = [
  'Phone',
  'WhatsApp',
  'SMS',
  'Email',
  'In person',
];

const AVAILABILITY_OPTIONS = [
  'Full-time',
  'Part-time',
  'A few hours per week',
  'Flexible / as needed',
];

export default function MarketerApplyScreen() {
  const [step, setStep] = useState(0);
  const [data, setData] = useState<ApplicationData>(INITIAL_DATA);
  const [submitting, setSubmitting] = useState(false);

  const { submitApplication } = useMarketers();

  const update = <K extends keyof ApplicationData>(
    key: K,
    value: ApplicationData[K],
  ) => {
    setData((current) => ({
      ...current,
      [key]: value,
    }));
  };

  const progress = useMemo(() => {
    return ((step + 1) / STEPS.length) * 100;
  }, [step]);

  const pickImageFromGallery = async (
    side: 'front' | 'back',
  ) => {
    const permission =
      await ImagePicker.requestMediaLibraryPermissionsAsync();

    if (!permission.granted) {
      Alert.alert(
        'Gallery permission required',
        'Please allow MAKAZII to access your photos so you can select your ID image.',
      );
      return;
    }

    const result = await ImagePicker.launchImageLibraryAsync({
      mediaTypes: ImagePicker.MediaTypeOptions.Images,
      allowsEditing: true,
      quality: 0.85,
      aspect: [4, 3],
    });

    if (result.canceled || !result.assets?.length) {
      return;
    }

    const uri = result.assets[0].uri;

    if (side === 'front') {
      update('idFront', uri);
    } else {
      update('idBack', uri);
    }
  };

  const takePicture = async (
    side: 'front' | 'back',
  ) => {
    const permission =
      await ImagePicker.requestCameraPermissionsAsync();

    if (!permission.granted) {
      Alert.alert(
        'Camera permission required',
        'Please allow MAKAZII to use your camera so you can take a picture of your ID.',
      );
      return;
    }

    const result = await ImagePicker.launchCameraAsync({
      mediaTypes: ImagePicker.MediaTypeOptions.Images,
      allowsEditing: true,
      quality: 0.85,
      aspect: [4, 3],
    });

    if (result.canceled || !result.assets?.length) {
      return;
    }

    const uri = result.assets[0].uri;

    if (side === 'front') {
      update('idFront', uri);
    } else {
      update('idBack', uri);
    }
  };

  const removeImage = (side: 'front' | 'back') => {
    if (side === 'front') {
      update('idFront', null);
    } else {
      update('idBack', null);
    }
  };

  const togglePlace = (place: string) => {
    setData((current) => {
      const exists = current.places.includes(place);

      return {
        ...current,
        places: exists
          ? current.places.filter((item) => item !== place)
          : [...current.places, place],
      };
    });
  };

  const toggleArrayValue = (
    key: 'sourcingMethods' | 'communicationMethods',
    value: string,
  ) => {
    setData((current) => {
      const currentValues = current[key];

      return {
        ...current,
        [key]: currentValues.includes(value)
          ? currentValues.filter((item) => item !== value)
          : [...currentValues, value],
      };
    });
  };

  const validateStep = (): boolean => {
    if (step === 0) {
      if (!data.firstName.trim()) {
        Alert.alert(
          'Missing information',
          'Enter your first name.',
        );
        return false;
      }

      if (!data.lastName.trim()) {
        Alert.alert(
          'Missing information',
          'Enter your last name.',
        );
        return false;
      }

      if (data.phone.trim().length < 9) {
        Alert.alert(
          'Invalid phone number',
          'Enter a valid phone number.',
        );
        return false;
      }

      if (!data.email.trim()) {
        Alert.alert(
          'Missing information',
          'Enter your email address.',
        );
        return false;
      }

      return true;
    }

    if (step === 1) {
      if (!data.idNumber.trim()) {
        Alert.alert(
          'Missing ID number',
          `Enter your ${
            data.idType === 'Passport'
              ? 'passport'
              : 'National ID'
          } number.`,
        );
        return false;
      }

      if (!data.idFront) {
        Alert.alert(
          'ID image required',
          `Please provide a picture of your ${
            data.idType === 'Passport'
              ? 'passport identification page'
              : 'ID front side'
          }.`,
        );
        return false;
      }

      if (
        data.idType === 'National ID' &&
        !data.idBack
      ) {
        Alert.alert(
          'ID back required',
          'Please provide a picture of the back side of your National ID.',
        );
        return false;
      }

      return true;
    }

    if (step === 2) {
      if (!data.county.trim()) {
        Alert.alert(
          'Missing county',
          'Enter or select your county.',
        );
        return false;
      }

      if (data.places.length === 0) {
        Alert.alert(
          'Select places',
          'Select at least one place you can actively cover.',
        );
        return false;
      }

      if (!data.timeInSelectedAreas) {
        Alert.alert(
          'Local experience',
          'Select how long you have lived or worked in your selected areas.',
        );
        return false;
      }

      return true;
    }

    if (step === 3) {
      if (!data.experience) {
        Alert.alert(
          'Experience',
          'Please tell us whether you have worked in real estate.',
        );
        return false;
      }

      if (data.experience === 'Yes') {
        if (!data.experienceType) {
          Alert.alert(
            'Experience area',
            'Please select the area of real estate you have worked in.',
          );
          return false;
        }

        if (!data.experienceLength) {
          Alert.alert(
            'Experience length',
            'Please select how long you have worked in real estate.',
          );
          return false;
        }
      }

      if (!data.occupation) {
        Alert.alert(
          'Current occupation',
          'Please select your current occupation.',
        );
        return false;
      }

      if (!data.motivation.trim()) {
        Alert.alert(
          'Missing answer',
          'Tell us why you want to become a MAKAZII marketer.',
        );
        return false;
      }

      if (!data.contribution.trim()) {
        Alert.alert(
          'Missing answer',
          'Tell us what you believe you can contribute to MAKAZII.',
        );
        return false;
      }

      if (!data.propertiesPerMonth) {
        Alert.alert(
          'Property sourcing',
          'Select how many properties you expect to source per month.',
        );
        return false;
      }

      if (!data.knowsLandlords) {
        Alert.alert(
          'Landlord network',
          'Please tell us whether you know landlords or property owners.',
        );
        return false;
      }

      if (!data.comfortableCommunication) {
        Alert.alert(
          'Customer interaction',
          'Please tell us whether you are comfortable communicating with renters and landlords.',
        );
        return false;
      }

      if (
        data.communicationMethods.length === 0
      ) {
        Alert.alert(
          'Communication methods',
          'Select at least one communication method.',
        );
        return false;
      }

      if (!data.availability) {
        Alert.alert(
          'Availability',
          'Select how much time you can dedicate.',
        );
        return false;
      }

      if (!data.followsProcedures) {
        Alert.alert(
          'MAKAZII procedures',
          'Please confirm whether you are willing to follow MAKAZII verification and customer-service procedures.',
        );
        return false;
      }

      return true;
    }

    if (step === 4) {
      if (!data.declaration) {
        Alert.alert(
          'Confirmation required',
          'Please confirm that the information provided is accurate and that the identification document belongs to you.',
        );
        return false;
      }

      return true;
    }

    return true;
  };

  const handleNext = () => {
    if (!validateStep()) {
      return;
    }

    if (step < STEPS.length - 1) {
      setStep((current) => current + 1);
      return;
    }

    submitApplicationForm();
  };

  const handleBack = () => {
    if (step === 0) {
      router.back();
      return;
    }

    setStep((current) => current - 1);
  };

  const submitApplicationForm = async () => {
    if (submitting) {
      return;
    }

    setSubmitting(true);

    try {
      const application = submitApplication({
        firstName: data.firstName.trim(),
        lastName: data.lastName.trim(),
        phone: data.phone.trim(),
        email: data.email.trim().toLowerCase(),

        idType: data.idType,
        idNumber: data.idNumber.trim(),

        idFront: data.idFront
          ? {
              uri: data.idFront,
              name:
                data.idType === 'Passport'
                  ? 'passport-identification-page.jpg'
                  : 'national-id-front.jpg',
              type: 'image',
            }
          : null,

        idBack: data.idBack
          ? {
              uri: data.idBack,
              name: 'national-id-back.jpg',
              type: 'image',
            }
          : null,

        county: data.county.trim(),
        countyId: data.countyId,
        constituencyId: data.constituencyId,
        wardIds: data.wardIds,
        placeIds: data.placeIds,
        places: data.places,
        timeInSelectedAreas:
          data.timeInSelectedAreas,

        hasRealEstateExperience:
          data.experience === 'Yes',

        experienceAreas: data.experienceType
          ? [data.experienceType]
          : [],

        experienceDuration:
          data.experienceLength,

        experienceDescription:
          data.experienceDescription.trim(),

        currentOccupation:
          data.occupation,

        workingWithAnotherAgency:
          data.workingWithAgency === 'Yes',

        currentAgencyName:
          data.agencyName.trim(),

        motivation:
          data.motivation.trim(),

        contribution:
          data.contribution.trim(),

        expectedPropertiesPerMonth:
          data.propertiesPerMonth,

        propertySources:
          data.sourcingMethods,

        knowsLandlords:
          data.knowsLandlords === 'Yes',

        approximateLandlordCount:
          data.landlordNetwork.trim()
            ? Number(data.landlordNetwork.trim())
            : null,

        comfortableCommunicating:
          data.comfortableCommunication === 'Yes',

        communicationMethods:
          data.communicationMethods,

        timeAvailability:
          data.availability,

        willingToFollowProcedures:
          data.followsProcedures === 'Yes',

        referenceName:
          data.referenceName.trim(),

        referencePhone:
          data.referencePhone.trim(),

        referenceRelationship:
          data.referenceRelationship.trim(),

        declarationAccepted:
          data.declaration,
      });

      router.replace({
        pathname: '/marketer/application',
        params: {
          id: application.id,
        },
      });
    } catch (error) {
      console.error(
        'Marketer application submission failed:',
        error,
      );

      Alert.alert(
        'Submission failed',
        'We could not submit your application right now. Please try again.',
      );
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <View style={styles.container}>
      <StatusBar barStyle="light-content" />

      <View style={styles.topBar}>
        <Pressable
          onPress={handleBack}
          style={({ pressed }) => [
            styles.backButton,
            pressed && styles.pressed,
          ]}
        >
          <ArrowLeft
            size={20}
            color={COLORS.white}
          />
        </Pressable>

        <View style={styles.brandWrap}>
          <Text style={styles.brand}>
            MAKAZII
          </Text>

          <Text style={styles.brandLabel}>
            MARKETER APPLICATION
          </Text>
        </View>

        <Text style={styles.stepCounter}>
          {step + 1}/{STEPS.length}
        </Text>
      </View>

      <View style={styles.progressArea}>
        <View style={styles.progressTrack}>
          <View
            style={[
              styles.progressFill,
              {
                width: `${progress}%`,
              },
            ]}
          />
        </View>

        <View style={styles.stepLabels}>
          {STEPS.map((item, index) => (
            <Text
              key={item}
              style={[
                styles.stepLabel,
                index === step &&
                  styles.stepLabelActive,
              ]}
            >
              {item}
            </Text>
          ))}
        </View>
      </View>

      <ScrollView
        contentContainerStyle={styles.content}
        keyboardShouldPersistTaps="handled"
        showsVerticalScrollIndicator={false}
      >
        {step === 0 && (
          <PersonalStep
            data={data}
            update={update}
          />
        )}

        {step === 1 && (
          <IdentityStep
            data={data}
            update={update}
            onTakePicture={takePicture}
            onPickGallery={pickImageFromGallery}
            onRemoveImage={removeImage}
          />
        )}

        {step === 2 && (
          <LocationStep
            data={data}
            update={update}
            togglePlace={togglePlace}
          />
        )}

        {step === 3 && (
          <ExperienceStep
            data={data}
            update={update}
            toggleArrayValue={toggleArrayValue}
          />
        )}

        {step === 4 && (
          <ReviewStep
            data={data}
            update={update}
          />
        )}
      </ScrollView>

      <View style={styles.bottomBar}>
        <Pressable
          onPress={handleNext}
          disabled={submitting}
          style={({ pressed }) => [
            styles.nextButton,
            pressed &&
              styles.nextButtonPressed,
            submitting &&
              styles.disabledButton,
          ]}
        >
          <Text style={styles.nextButtonText}>
            {submitting
              ? 'Submitting...'
              : step ===
                  STEPS.length - 1
                ? 'Submit Application'
                : 'Continue'}
          </Text>

          {step <
            STEPS.length - 1 &&
          !submitting ? (
            <ArrowRight
              size={20}
              color={COLORS.background}
              strokeWidth={2.5}
            />
          ) : (
            <Check
              size={20}
              color={COLORS.background}
              strokeWidth={2.5}
            />
          )}
        </Pressable>
      </View>
    </View>
  );
}

/* -------------------------------------------------------------------------- */
/* PERSONAL                                                                    */
/* -------------------------------------------------------------------------- */

function PersonalStep({
  data,
  update,
}: {
  data: ApplicationData;
  update: <K extends keyof ApplicationData>(
    key: K,
    value: ApplicationData[K],
  ) => void;
}) {
  return (
    <View>
      <StepHeader
        icon={
          <UserRound
            size={22}
            color={COLORS.lime}
          />
        }
        eyebrow="STEP 1"
        title="Tell us about yourself"
        description="Start with the basic information we need to identify and contact you."
      />

      <Field
        label="First name"
        value={data.firstName}
        onChangeText={(value) =>
          update('firstName', value)
        }
        placeholder="e.g. Brian"
      />

      <Field
        label="Last name"
        value={data.lastName}
        onChangeText={(value) =>
          update('lastName', value)
        }
        placeholder="e.g. Kamau"
      />

      <Field
        label="Phone number"
        value={data.phone}
        onChangeText={(value) =>
          update('phone', value)
        }
        placeholder="e.g. 0712 345 678"
        keyboardType="phone-pad"
      />

      <Field
        label="Email address"
        value={data.email}
        onChangeText={(value) =>
          update('email', value)
        }
        placeholder="e.g. brian@example.com"
        keyboardType="email-address"
        autoCapitalize="none"
      />

      <InfoCard
        icon={
          <ShieldCheck
            size={19}
            color={COLORS.lime}
          />
        }
        title="Your information"
        text="Your application information will be used for MAKAZII's review process. Sensitive identity documents will only be accessible to authorized administrators."
      />
    </View>
  );
}

/* -------------------------------------------------------------------------- */
/* IDENTITY                                                                    */
/* -------------------------------------------------------------------------- */

function IdentityStep({
  data,
  update,
  onTakePicture,
  onPickGallery,
  onRemoveImage,
}: {
  data: ApplicationData;
  update: <K extends keyof ApplicationData>(
    key: K,
    value: ApplicationData[K],
  ) => void;
  onTakePicture: (
    side: 'front' | 'back',
  ) => void;
  onPickGallery: (
    side: 'front' | 'back',
  ) => void;
  onRemoveImage: (
    side: 'front' | 'back',
  ) => void;
}) {
  return (
    <View>
      <StepHeader
        icon={
          <ShieldCheck
            size={22}
            color={COLORS.lime}
          />
        }
        eyebrow="STEP 2"
        title="Verify your identity"
        description="Provide a clear picture of your identification document. You can take a new picture or choose one from your gallery."
      />

      <Text style={styles.fieldLabel}>
        Identification type
      </Text>

      <View style={styles.choiceRow}>
        <ChoiceButton
          label="National ID"
          selected={
            data.idType === 'National ID'
          }
          onPress={() => {
            update(
              'idType',
              'National ID',
            );
            update('idFront', null);
            update('idBack', null);
          }}
        />

        <ChoiceButton
          label="Passport"
          selected={
            data.idType === 'Passport'
          }
          onPress={() => {
            update(
              'idType',
              'Passport',
            );
            update('idFront', null);
            update('idBack', null);
          }}
        />
      </View>

      <Field
        label={
          data.idType === 'Passport'
            ? 'Passport number'
            : 'National ID number'
        }
        value={data.idNumber}
        onChangeText={(value) =>
          update('idNumber', value)
        }
        placeholder={
          data.idType === 'Passport'
            ? 'Enter passport number'
            : 'Enter ID number'
        }
      />

      <DocumentUpload
        title={
          data.idType === 'Passport'
            ? 'Passport identification page'
            : 'National ID — front side'
        }
        image={data.idFront}
        onTakePicture={() =>
          onTakePicture('front')
        }
        onPickGallery={() =>
          onPickGallery('front')
        }
        onRemove={() =>
          onRemoveImage('front')
        }
      />

      {data.idType ===
        'National ID' && (
        <DocumentUpload
          title="National ID — back side"
          image={data.idBack}
          onTakePicture={() =>
            onTakePicture('back')
          }
          onPickGallery={() =>
            onPickGallery('back')
          }
          onRemove={() =>
            onRemoveImage('back')
          }
        />
      )}

      <InfoCard
        icon={
          <ShieldCheck
            size={19}
            color={COLORS.lime}
          />
        }
        title="Keep your document clear"
        text="Make sure all text and numbers are readable and the entire document is visible. Your identity images are private application documents."
      />
    </View>
  );
}

/* -------------------------------------------------------------------------- */
/* LOCATION                                                                    */
/* -------------------------------------------------------------------------- */

function LocationStep({
  data,
  update,
  togglePlace,
}: {
  data: ApplicationData;
  update: <K extends keyof ApplicationData>(
    key: K,
    value: ApplicationData[K],
  ) => void;
  togglePlace: (place: string) => void;
}) {
  return (
    <View>
      <StepHeader
        icon={
          <MapPin
            size={22}
            color={COLORS.lime}
          />
        }
        eyebrow="STEP 3"
        title="Where can you work?"
        description="Choose the county and places where you have local knowledge and can actively source properties."
      />

      <Field
        label="County"
        value={data.county}
        onChangeText={(value) =>
          update('county', value)
        }
        placeholder="e.g. Nairobi"
      />

      <Text style={styles.fieldLabel}>
        Places you can cover
      </Text>

      <Text style={styles.helperText}>
        Select all places where you have
        local knowledge or connections.
      </Text>

      <View style={styles.chipWrap}>
        {PLACE_OPTIONS.map((place) => {
          const selected =
            data.places.includes(place);

          return (
            <Pressable
              key={place}
              onPress={() =>
                togglePlace(place)
              }
              style={[
                styles.chip,
                selected &&
                  styles.chipSelected,
              ]}
            >
              {selected && (
                <Check
                  size={13}
                  color={COLORS.background}
                  strokeWidth={3}
                />
              )}

              <Text
                style={[
